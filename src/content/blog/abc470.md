[Monocolor ](https://atcoder.jp/contests/abc470/tasks/abc470_b)
```Haskell
import Data.List (group, sort)
import Control.Monad (forM)

main :: IO ()
main = do
  n <- readLn :: IO Int
  c_l <- map read . words <$> getLine :: IO [Int]

  print $ length c_l - maximum (elemCountOrd c_l)

elemCountOrd :: (Ord a) => [a] -> [Int]
elemCountOrd = map (\l -> length l ) . group . sort
```
