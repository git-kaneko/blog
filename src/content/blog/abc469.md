[Isolated Seats](https://atcoder.jp/contests/abc469/tasks/abc469_b)

```Haskell
main :: IO ()
main = do
  n <- readLn
  as <- getLine

  print $ solve n as

solve :: Int -> [Char] -> Int
solve 1 as = if head as == 'x' then 1 else 0
solve n as = do
  let x = length [i | i <- [0..n-3], as !! i == 'x', as !! (i+1) =='x', as !! (i+2) =='x']
  let right = if (take 2 as) == "xx" then 1 else 0
  let left = if as !! (n-1) == 'x' && as !! (n-2) == 'x' then 1 else 0

  x + right + left
```
